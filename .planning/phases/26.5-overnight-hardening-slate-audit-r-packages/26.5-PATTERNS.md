**Core pattern (follow BabystepPrefs exactly):**
```kotlin
class DisplayPrefs(
    private val dataStore: DataStore<Preferences>,
) {
    /**
     * Whether to keep the screen awake while the app is in the foreground.
     * Default TRUE — the dedicated-display use case (R2 §1). Fail-safe: a read
     * error yields the default (never crashes).
     *
     * WRITE-SCOPE CONTRACT ([[dinghy-compose-write-scope-cancellation]]): the UI
     * NEVER calls [setKeepScreenOn] from a composition scope. Route through
     * [AppContainer.setKeepScreenOn] (writeScope) — same pattern as [BabystepPrefs].
     */
    val keepScreenOn: Flow<Boolean> =
        dataStore.data
            .catch { e -> if (e is IOException) emit(emptyPreferences()) else throw e }
            .map { prefs -> prefs[KEY_KEEP_SCREEN_ON] ?: DEFAULT_KEEP_SCREEN_ON }

    /** Persist the keep-screen-on toggle. */
    suspend fun setKeepScreenOn(on: Boolean) {
        dataStore.edit { prefs -> prefs[KEY_KEEP_SCREEN_ON] = on }
    }

    companion object {
        const val DEFAULT_KEEP_SCREEN_ON = true
        private val KEY_KEEP_SCREEN_ON = booleanPreferencesKey("keep_screen_on")
    }
}
```
